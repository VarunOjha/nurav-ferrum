// SPIKE S1 (IP-1, temporary; deleted after the result is recorded).
// Reads ../wiki/SCHEMA.md and writes + deletes ../plan/.spike-<rand> at request time.
import { createHash, randomBytes } from "node:crypto";
import { access, mkdir, readFile, rm, rmdir, stat, writeFile } from "node:fs/promises";
import path from "node:path";

export const runtime = "nodejs";
export const dynamic = "force-dynamic";

export async function GET() {
  const root = path.resolve(process.cwd(), "..");
  const schemaPath = path.join(root, "wiki", "SCHEMA.md");
  const planDir = path.join(root, "plan");
  const approvals = path.join(planDir, "approvals.json");
  const result: Record<string, unknown> = { cwd: process.cwd(), root };
  try {
    const buf = await readFile(schemaPath);
    result.schemaBytes = buf.length;
    result.schemaSha256 = createHash("sha256").update(buf).digest("hex");
    result.schemaFirstLine = buf.toString("utf8").split("\n")[0];

    let createdPlanDir = false;
    try {
      await stat(planDir);
    } catch {
      await mkdir(planDir);
      createdPlanDir = true;
    }
    const spike = path.join(planDir, `.spike-${randomBytes(6).toString("hex")}`);
    await writeFile(spike, "spike S1\n");
    result.spikeWritten = path.basename(spike);
    result.spikeReadBack = (await readFile(spike, "utf8")).trim();
    await rm(spike);
    let stillThere = true;
    try {
      await access(spike);
    } catch {
      stillThere = false;
    }
    result.spikeDeleted = !stillThere;
    if (createdPlanDir) await rmdir(planDir);
    result.createdAndRemovedPlanDir = createdPlanDir;
    let approvalsExists = true;
    try {
      await access(approvals);
    } catch {
      approvalsExists = false;
    }
    result.approvalsJsonExists = approvalsExists;
    result.ok = true;
  } catch (e) {
    result.ok = false;
    result.error = String(e);
  }
  return Response.json(result);
}

"use client";
// SPIKE S2 (IP-1, temporary; deleted after the result is recorded).
import { useEffect, useState } from "react";

const SOURCE = `flowchart LR
  ip_spec["ip-spec"] --> ip_build_plan["ip-build-plan"]
  ip_build_plan -.->|TB| schema["SCHEMA"]`;

export default function MermaidSpike() {
  const [svg, setSvg] = useState("");
  const [state, setState] = useState("pending");
  useEffect(() => {
    let cancelled = false;
    (async () => {
      try {
        const mermaid = (await import("mermaid")).default;
        mermaid.initialize({
          startOnLoad: false,
          securityLevel: "strict",
          layout: "dagre",
          look: "classic",
        });
        const out = await mermaid.render("spike-s2", SOURCE);
        if (!cancelled) {
          setSvg(out.svg);
          setState("rendered");
        }
      } catch (e) {
        if (!cancelled) setState(`error: ${String(e)}`);
      }
    })();
    return () => {
      cancelled = true;
    };
  }, []);
  return (
    <div>
      <p data-testid="state">{state}</p>
      <div data-testid="diagram" dangerouslySetInnerHTML={{ __html: svg }} />
    </div>
  );
}
